(() => {
 'use strict';
 const views=[...document.querySelectorAll('.view')],tabs=document.querySelector('.tabs');let currentView='feed',activeDemoUser={name:'Riya Creates',handle:'@riya_demo'},exitWarningAt=0;
 function renderView(id){if(!document.getElementById(id))id='feed';currentView=id;views.forEach(v=>v.classList.toggle('active',v.id===id));document.querySelectorAll('.tab[data-view]').forEach(b=>b.classList.toggle('active',b.dataset.view===id));if(tabs)tabs.hidden=['account','login','settings'].includes(id);document.body.classList.toggle('registration-open',['account','login'].includes(id));window.scrollTo(0,0);}
 function navigate(id){if(!document.getElementById(id)||id===currentView)return;history.pushState({yunex:true,view:id},'','#'+id);renderView(id);}
 history.replaceState({yunex:true,view:'feed'},'',location.pathname+location.search);renderView('feed');
 document.addEventListener('click',e=>{
  const p=e.target.closest('.profile-open');if(p){activeDemoUser={name:p.dataset.user||'YUNEX Creator',handle:p.dataset.handle||'@creator_demo'};document.getElementById('demoName').textContent=activeDemoUser.name;document.getElementById('demoHandle').textContent=activeDemoUser.handle;document.getElementById('demoAvatar').textContent=activeDemoUser.name.split(/\s+/).map(x=>x[0]).join('').slice(0,2).toUpperCase();document.getElementById('chatName').textContent=activeDemoUser.name;document.getElementById('chatAvatar').textContent=activeDemoUser.name[0].toUpperCase();navigate('demoProfile');return;}
  const vb=e.target.closest('[data-view]');if(vb){e.preventDefault();navigate(vb.dataset.view);return;}
  if(e.target.closest('#profileShortcut')){navigate('profile');return;}
  if(e.target.closest('.brand')){e.preventDefault();navigate('feed');return;}
  if(e.target.closest('#followDemo')){const b=document.getElementById('followDemo'),on=b.classList.toggle('following');b.textContent=on?'Following':'Follow';return;}
  if(e.target.closest('#messageDemo')){navigate('chat');return;}
  const like=e.target.closest('.like-btn');if(like){const on=like.classList.toggle('liked');like.firstChild.textContent=on?'♥ ':'♡ ';const label=like.querySelector('span');if(label)label.textContent=on?'Liked':'Like';return;}
  const demo=e.target.closest('[data-demo]');if(demo){const msg=demo.dataset.demo==='forgot'?'Password recovery is not enabled yet. It will be added with a verified recovery channel to protect accounts.':demo.dataset.demo==='comment'?'Comments are not connected in this demo.':'Sharing will be available after backend integration.';alert(msg);}
 });
 window.addEventListener('popstate',e=>{if(e.state?.yunex&&e.state.view){renderView(e.state.view);return;}if(currentView==='feed'){const now=Date.now();if(exitWarningAt&&now-exitWarningAt<=2000){exitWarningAt=0;return;}exitWarningAt=now;history.pushState({yunex:true,view:'feed'},'','#feed');let toast=document.getElementById('exitToast');if(!toast){toast=document.createElement('div');toast.id='exitToast';toast.className='exit-toast';toast.textContent='Press back again to exit';document.body.appendChild(toast);}toast.classList.add('show');setTimeout(()=>toast.classList.remove('show'),1800);}else renderView('feed');});
 const fileInput=document.getElementById('mediaFile');fileInput?.addEventListener('change',()=>{const file=fileInput.files[0],preview=document.getElementById('preview');preview.replaceChildren();if(!file)return;const el=document.createElement(file.type.startsWith('video/')?'video':'img');el.src=URL.createObjectURL(file);if(el.tagName==='VIDEO')el.controls=true;preview.appendChild(el);});
 document.getElementById('demoPost')?.addEventListener('click',()=>{const m=document.getElementById('uploadMessage');m.hidden=false;m.textContent=fileInput.files.length?'Local preview ready. File not uploaded or published.':'Choose a photo or video first.';});
 document.querySelectorAll('.category-grid button').forEach(b=>b.addEventListener('click',()=>{document.getElementById('exploreSearch').value=b.innerText.replace(/[^\w\s]/g,'').trim();}));
 document.getElementById('chatForm')?.addEventListener('submit',e=>{e.preventDefault();const input=document.getElementById('chatInput'),v=input.value.trim();if(!v)return;const bubble=document.createElement('div');bubble.className='chat-bubble sent';bubble.textContent=v;document.getElementById('chatMessages').appendChild(bubble);input.value='';bubble.scrollIntoView({behavior:'smooth',block:'nearest'});});
 const theme=document.getElementById('themeSelect');if(theme){const saved=localStorage.getItem('yunex-demo-theme')||'dark';theme.value=saved;document.body.classList.toggle('light-theme',saved==='light');theme.addEventListener('change',()=>{document.body.classList.toggle('light-theme',theme.value==='light');localStorage.setItem('yunex-demo-theme',theme.value);});}
})();
